from itertools import groupby

data = [
    {'id': 1, 'name': 'Alice', 'city': 'New York'},
    {'id': 3, 'name': 'Charlie', 'city': 'New York'},
    {'id': 2, 'name': 'Bob', 'city': 'London'},
    {'id': 4, 'name': 'David', 'city': 'London'}
]

# Sort the data by the grouping key ('city')
data.sort(key=lambda x: x['city'])

grouped_by_city_itertools = {
    key: list(group) for key, group in groupby(data, key=lambda x: x['city'])
}

print(grouped_by_city_itertools)

import pandas as pd

# Create a sample DataFrame
data = {'Category': ['A', 'B', 'A', 'C', 'B', 'A'],
        'Value': [10, 20, 15, 30, 25, 12]}
df = pd.DataFrame(data)

# Group by 'Category' and calculate the sum of 'Value'
grouped_data = df.groupby('Category')['Value'].sum()
print(grouped_data)

# Apply multiple aggregations to different columns
aggregated_data = df.groupby('Category').agg(
    Total_Value=('Value', 'sum'),
    Mean_Value=('Value', 'mean')
)
print(aggregated_data)
